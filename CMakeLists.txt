    list(SORT _MVVM_LLVM_CONFIG_DIRS COMPARE NATURAL ORDER DESCENDING)
    foreach (_MVVM_LLVM_CONFIG_DIR IN LISTS _MVVM_LLVM_CONFIG_DIRS)
        get_filename_component(_MVVM_LLVM_ROOT "${_MVVM_LLVM_CONFIG_DIR}/../../.." ABSOLUTE)
        if (EXISTS "${_MVVM_LLVM_ROOT}/bin/llvm-tblgen")
            set(LLVM_DIR "${_MVVM_LLVM_CONFIG_DIR}" CACHE PATH "LLVM CMake package directory")
            break()
        endif ()
    endforeach ()
endif ()
